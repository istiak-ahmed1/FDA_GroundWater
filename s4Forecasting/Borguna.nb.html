library(ftsa)
df <- read.csv("Borguna.csv")
y_mat <- as.matrix(df)

colnames(y_mat) <- as.character(2015:2024)

x <- 1:nrow(y_mat)        # Months (1–12)
time <- 1:ncol(y_mat)     # Years (10)
fts_data <- fts(
  x = x,
  y = y_mat,
  xname = "Month",
  yname = "Water Table Depth"
)
fit <- ftsm(fts_data, order = 3, method = 'rapca')
fit$varprop
[1] 0.5596575 0.1659240 0.1028485
fc <- forecast(fit, h = 6, method = 'arfima', level=c(95))
